import random

def get_action(cart_position, cart_velocity, pole_angle, pole_angular_velocity):
    action = None
    if cart_position > 20:
        if pole_angle + 0.5 * pole_angular_velocity > 0:
            action = 2
        else:
            action = 1
    elif cart_position < -20:
        if pole_angle + 0.5 * pole_angular_velocity < 0:
            action = 1
        else:
            action = 2
    else:
        if abs(pole_angle) > 12:
            if pole_angle > 0:
                action = 2
            else:
                action = 1
        else:
            if pole_angle + 0.5 * pole_angular_velocity > 0:
                action = 2
            else:
                action = 1
    if action is None:
        return random.randint(1, 2)
    return action

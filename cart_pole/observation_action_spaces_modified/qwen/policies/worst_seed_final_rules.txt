def balance_pole(cart_pos, cart_vel, pole_ang, pole_vel):
    if abs(pole_ang) > 15:
        return 1 if pole_ang < 0 else 2
    elif abs(cart_pos) > 24:
        return 1 if cart_pos < 0 else 2
    elif abs(pole_vel) > 5:
        return 1 if pole_vel < 0 else 2
    elif abs(cart_vel) > 5:
        return 1 if cart_vel < 0 else 2
    elif abs(pole_ang) > 3 or abs(pole_vel) > 2:
        return 1 if pole_ang < -3 or pole_vel < -2 else 2
    elif cart_vel > 0.5:
        return 2
    elif cart_vel < -0.5:
        return 1
    else:
        return 1

def choose_action(observation):
    position, velocity = observation

    if position < -0.5:
        return 0.3
    elif position > 0.3:
        return -0.3
    else:
        if velocity < -0.1:
            return 0.2 + 0.1 * velocity
        elif velocity > 0.1:
            return -0.2 - 0.1 * velocity
        else:
            return 0.1 * velocity

    return 0.0

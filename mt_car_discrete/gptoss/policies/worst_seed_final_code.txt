import math, random

def get_action(car_position, car_velocity):
    GOAL_POS = 0.5
    GRAVITY = 0.0025

    def v_needed(position):
        delta = math.sin(3 * GOAL_POS) - math.sin(3 * position)
        if delta <= 0:
            return 0.0
        return math.sqrt(2 * GRAVITY * delta)

    if car_position >= GOAL_POS:
        return 1
    elif car_position < 0:
        if car_velocity < 0:
            return 0
        elif car_velocity < v_needed(car_position):
            return 2
        else:
            return 1
    else:
        if car_velocity < v_needed(car_position):
            return 2
        else:
            return 1

    return random.randint(0, 2)

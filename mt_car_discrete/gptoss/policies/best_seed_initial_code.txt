def get_action(car_position, car_velocity):
    import random
    p = car_position
    v = car_velocity
    if p < 0 and v < 0.2:
        return 2
    elif p < 0 and 0.2 <= v < 0.5:
        return 1
    elif p < 0 and v >= 0.5:
        return 0
    elif 0 <= p < 0.3 and v < 0.4:
        return 2
    elif 0 <= p < 0.3 and 0.4 <= v < 0.8:
        return 1
    elif 0 <= p < 0.3 and v >= 0.8:
        return 0
    elif p >= 0.3 and v < 0:
        return 2
    elif p >= 0.3 and v >= 0:
        return 1
    else:
        return random.randint(0, 2)
